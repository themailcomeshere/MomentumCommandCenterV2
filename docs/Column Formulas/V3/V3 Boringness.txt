# ============================================================
# MCC V3 BORINGNESS  (version MCC-V3-BORING-0.1, score map BORING-0.1-PROVISIONAL)
# Thinkorswim custom watchlist column.
# SETTINGS: Aggregation = D (daily). Extended hours OFF.
# C# reference: BoringOvernightV3Calculator (BoringnessScore, BoringnessBand).
# STATUS: Not yet compiled in Thinkorswim. Repository text is not proof of compilation.
#
# VALUE (sortable): Boringness Score 0-100, kept separate from the
#   Overnight Edge Score. Point maps from
#   docs/project-control/Boring_Overnight_Column_Data_Contract_v1.md 10A
#   (provisional, not calibrated). Linear interpolation between anchors:
#   Win rate     25: 60%=0, 70%=10, 80%=18, 90%+=25
#   Median ON    20: <=0%=0, 0.10%=8, 0.25%=14, 0.50%+=20
#   Mean ON 20D  15: <0.10%=0, 0.10%=5, 0.25%=10, 0.50%=15, 1.00%=10, 1.50%=5, 2.00%+=0
#   5D - 20D     10: <=0=0, just above 0=5, 0.10pp=7, 0.50pp=10, 1.00pp=7, 3.00pp=3, 5.00pp+=0
#   Consistency  10: share of 20 returns in [0%, +2%]: 60%=0, 70%=5, 80%=8, 90%+=10
#   Volatility   10: population std dev of 20 returns: <=0.25%=10, 0.50%=8, 1.00%=5, 1.50%=2, 2.00%+=0
#   Avg volume    5: prior 20 sessions: <500k=0, 500k=1, 750k=2, 1M=3, 2M+=5
#   Price         5: $5=0, $10=1, $25=3, $50+=5
# TEXT: score + band. 90+ PERFECT, 80+ VERY GOOD, 70+ POSSIBLE,
#   60+ AGGRESSIVE, below 60 REJECT. A band never overrides a gate.
# Volatility is written out as sqrt(mean(r^2) - mean(r)^2) so it does not
# depend on Thinkorswim's StDev definition.
# ============================================================

script Lerp {
    input x = 0.0;
    input x0 = 0.0;
    input y0 = 0.0;
    input x1 = 1.0;
    input y1 = 0.0;
    plot value = y0 + (y1 - y0) * (x - x0) / (x1 - x0);
}

def r = open / close[1] - 1;
def enoughHistory = !IsNaN(close[20]) and !IsNaN(volume[20]);

def win20 = Average(if r > 0 then 1 else 0, 20);
def avg20 = Average(r, 20);
def median20 = Median(r, 20);
def avg5 = Average(r, 5);

def winPct = win20 * 100;
def medianPct = median20 * 100;
def meanPct = avg20 * 100;
def diffPp = (avg5 - avg20) * 100;
def consistencyPct = Average(if r >= 0 and r <= 0.02 then 1 else 0, 20) * 100;
def volatilityPct = Sqrt(Max(0, Average(r * r, 20) - avg20 * avg20)) * 100;
def avgVolumePrior20 = Average(volume[1], 20);
def price = close;

def ptsWin =
    if winPct <= 60 then 0
    else if winPct <= 70 then Lerp(winPct, 60, 0, 70, 10)
    else if winPct <= 80 then Lerp(winPct, 70, 10, 80, 18)
    else if winPct <= 90 then Lerp(winPct, 80, 18, 90, 25)
    else 25;

def ptsMedian =
    if medianPct <= 0 then 0
    else if medianPct <= 0.10 then Lerp(medianPct, 0, 0, 0.10, 8)
    else if medianPct <= 0.25 then Lerp(medianPct, 0.10, 8, 0.25, 14)
    else if medianPct <= 0.50 then Lerp(medianPct, 0.25, 14, 0.50, 20)
    else 20;

def ptsMean =
    if meanPct < 0.10 then 0
    else if meanPct <= 0.25 then Lerp(meanPct, 0.10, 5, 0.25, 10)
    else if meanPct <= 0.50 then Lerp(meanPct, 0.25, 10, 0.50, 15)
    else if meanPct <= 1.00 then Lerp(meanPct, 0.50, 15, 1.00, 10)
    else if meanPct <= 1.50 then Lerp(meanPct, 1.00, 10, 1.50, 5)
    else if meanPct <= 2.00 then Lerp(meanPct, 1.50, 5, 2.00, 0)
    else 0;

def ptsFiveDay =
    if diffPp <= 0 then 0
    else if diffPp <= 0.10 then Lerp(diffPp, 0, 5, 0.10, 7)
    else if diffPp <= 0.50 then Lerp(diffPp, 0.10, 7, 0.50, 10)
    else if diffPp <= 1.00 then Lerp(diffPp, 0.50, 10, 1.00, 7)
    else if diffPp <= 3.00 then Lerp(diffPp, 1.00, 7, 3.00, 3)
    else if diffPp <= 5.00 then Lerp(diffPp, 3.00, 3, 5.00, 0)
    else 0;

def ptsConsistency =
    if consistencyPct <= 60 then 0
    else if consistencyPct <= 70 then Lerp(consistencyPct, 60, 0, 70, 5)
    else if consistencyPct <= 80 then Lerp(consistencyPct, 70, 5, 80, 8)
    else if consistencyPct <= 90 then Lerp(consistencyPct, 80, 8, 90, 10)
    else 10;

def ptsVolatility =
    if volatilityPct <= 0.25 then 10
    else if volatilityPct <= 0.50 then Lerp(volatilityPct, 0.25, 10, 0.50, 8)
    else if volatilityPct <= 1.00 then Lerp(volatilityPct, 0.50, 8, 1.00, 5)
    else if volatilityPct <= 1.50 then Lerp(volatilityPct, 1.00, 5, 1.50, 2)
    else if volatilityPct <= 2.00 then Lerp(volatilityPct, 1.50, 2, 2.00, 0)
    else 0;

def ptsVolume =
    if avgVolumePrior20 < 500000 then 0
    else if avgVolumePrior20 <= 750000 then Lerp(avgVolumePrior20, 500000, 1, 750000, 2)
    else if avgVolumePrior20 <= 1000000 then Lerp(avgVolumePrior20, 750000, 2, 1000000, 3)
    else if avgVolumePrior20 <= 2000000 then Lerp(avgVolumePrior20, 1000000, 3, 2000000, 5)
    else 5;

def ptsPrice =
    if price <= 5 then 0
    else if price <= 10 then Lerp(price, 5, 0, 10, 1)
    else if price <= 25 then Lerp(price, 10, 1, 25, 3)
    else if price <= 50 then Lerp(price, 25, 3, 50, 5)
    else 5;

def boringness = ptsWin + ptsMedian + ptsMean + ptsFiveDay
    + ptsConsistency + ptsVolatility + ptsVolume + ptsPrice;

plot BoringnessScore = if enoughHistory then boringness else Double.NaN;

AddLabel(yes,
    if !enoughHistory then "DATA? <21 BARS"
    else AsText(boringness, NumberFormat.TWO_DECIMAL_PLACES) + " " +
        (if boringness >= 90 then "PERFECT"
         else if boringness >= 80 then "VERY GOOD"
         else if boringness >= 70 then "POSSIBLE"
         else if boringness >= 60 then "AGGRESSIVE"
         else "REJECT"),
    Color.BLACK);

AssignBackgroundColor(
    if !enoughHistory then Color.GRAY
    else if boringness >= 80 then Color.GREEN
    else if boringness >= 70 then CreateColor(150, 220, 120)
    else if boringness >= 60 then Color.YELLOW
    else Color.LIGHT_RED);
