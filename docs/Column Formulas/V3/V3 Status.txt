# ============================================================
# MCC V3 STATUS  (version MCC-V3-BORING-0.1)
# Thinkorswim custom watchlist column.
# SETTINGS: Aggregation = D (daily). Extended hours OFF.
# C# reference: BoringOvernightV3Calculator (Status, StatusCode, Reasons).
# STATUS: Not yet compiled in Thinkorswim. Repository text is not proof of compilation.
#
# VALUE (sortable): 3 CANDIDATE, 2 WATCH, 1 BLOCKED, 0 DATA?
# TEXT: status + primary reason (+n = further reasons; MCC lists them all).
#
# This column NEVER shows READY. CANDIDATE means every Thinkorswim-computable
# V3 gate passed; the symbol still needs MCC's READY check (catalyst sign-off,
# exclusion list, data freshness, 5M permission, buy window). Not an order.
#
# BLOCKED (hard gates, checked in this order):
#   PRICE<5      close below $5.00 (exactly $5.00 passes; operator decision 2026-11-16)
#   AVGVOL<500K  average volume of the prior 20 sessions below 500,000
#   WIN<60       fewer than 60% of the last 20 overnight returns were > 0
#   AVG<0.10     20-session mean overnight return below +0.10%
#   5D<=20D      5-session mean not strictly above the 20-session mean
#   BELOW SMA20  close not above the 20-session simple average close
# WATCH (current-day cautions, checked in this order):
#   ON>=3%       today's overnight return (open vs prior close) is 3% or more
#   5D>=5%       5-session mean overnight return is 5% or more
#   5D JUMP      5D mean minus 20D mean is 3 percentage points or more
#   RANGE>1.5ATR today's high-low range / prior close exceeds 1.5 x ATR%
#                (ATR = simple mean of the prior 14 true ranges / prior close)
#   VOL>=3X      today's volume is 3x or more the prior 20-session average
#                (during the session today's volume is partial, so this
#                 only triggers on clearly abnormal volume)
#   BORING<60    Boringness Score below 60 (same formula as V3 Boringness)
# DATA?: fewer than 21 daily bars loaded.
# ============================================================

def r = open / close[1] - 1;
def enoughHistory = !IsNaN(close[20]) and !IsNaN(volume[20]);

# ---------- Edge metrics (same as V3 Edge) ----------
def win20 = Average(if r > 0 then 1 else 0, 20);
def avg20 = Average(r, 20);
def median20 = Median(r, 20);
def avg5 = Average(r, 5);
def aboveSma20 = close > Average(close, 20);
def avgVolumePrior20 = Average(volume[1], 20);

# ---------- Boringness (same as V3 Boringness) ----------
def winPct = win20 * 100;
def medianPct = median20 * 100;
def meanPct = avg20 * 100;
def diffPp = (avg5 - avg20) * 100;
def consistencyPct = Average(if r >= 0 and r <= 0.02 then 1 else 0, 20) * 100;
def volatilityPct = Sqrt(Max(0, Average(r * r, 20) - avg20 * avg20)) * 100;
def price = close;

# Point maps as clamped segment sums: no helper script and no if-chains, so the
# column stays under Thinkorswim's real-time complexity limit. Each term adds
# slope * (portion of x inside that segment). Same values as the C# reference.
def ptsWin =
    1.0 * (Max(60, Min(70, winPct)) - 60)
    + 0.8 * (Max(70, Min(80, winPct)) - 70)
    + 0.7 * (Max(80, Min(90, winPct)) - 80);

def ptsMedian =
    80 * (Max(0, Min(0.10, medianPct)) - 0)
    + 40 * (Max(0.10, Min(0.25, medianPct)) - 0.10)
    + 24 * (Max(0.25, Min(0.50, medianPct)) - 0.25);

def ptsMean =
    if meanPct < 0.10 then 0
    else 5
    + (5 / 0.15) * (Max(0.10, Min(0.25, meanPct)) - 0.10)
    + 20 * (Max(0.25, Min(0.50, meanPct)) - 0.25)
    - 10 * (Max(0.50, Min(2.00, meanPct)) - 0.50);

def ptsFiveDay =
    if diffPp <= 0 then 0
    else 5
    + 20 * (Max(0, Min(0.10, diffPp)) - 0)
    + 7.5 * (Max(0.10, Min(0.50, diffPp)) - 0.10)
    - 6 * (Max(0.50, Min(1.00, diffPp)) - 0.50)
    - 2 * (Max(1.00, Min(3.00, diffPp)) - 1.00)
    - 1.5 * (Max(3.00, Min(5.00, diffPp)) - 3.00);

def ptsConsistency =
    0.5 * (Max(60, Min(70, consistencyPct)) - 60)
    + 0.3 * (Max(70, Min(80, consistencyPct)) - 70)
    + 0.2 * (Max(80, Min(90, consistencyPct)) - 80);

def ptsVolatility =
    10
    - 8 * (Max(0.25, Min(0.50, volatilityPct)) - 0.25)
    - 6 * (Max(0.50, Min(1.50, volatilityPct)) - 0.50)
    - 4 * (Max(1.50, Min(2.00, volatilityPct)) - 1.50);

def ptsVolume =
    if avgVolumePrior20 < 500000 then 0
    else 1
    + (Max(500000, Min(1000000, avgVolumePrior20)) - 500000) / 250000
    + (Max(1000000, Min(2000000, avgVolumePrior20)) - 1000000) / 500000;

def ptsPrice =
    0.2 * (Max(5, Min(10, price)) - 5)
    + (2 / 15) * (Max(10, Min(25, price)) - 10)
    + 0.08 * (Max(25, Min(50, price)) - 25);

def boringness = ptsWin + ptsMedian + ptsMean + ptsFiveDay
    + ptsConsistency + ptsVolatility + ptsVolume + ptsPrice;

# ---------- Current-day inputs ----------
def atrPct = Average(TrueRange(high, close, low)[1], 14) / close[1] * 100;
def rangePct = (high - low) / close[1] * 100;
def volumeRatio = volume / avgVolumePrior20;

# ---------- Gates and cautions ----------
def gPrice = close < 5;
def gVolume = avgVolumePrior20 < 500000;
def gWin = win20 < 0.60;
def gMean = avg20 < 0.001;
def gFiveDay = avg5 <= avg20;
def gSma = !aboveSma20;
def blockedCount = gPrice + gVolume + gWin + gMean + gFiveDay + gSma;

def wOn = r * 100 >= 3;
def wFiveDayHigh = avg5 * 100 >= 5;
def wJump = diffPp >= 3;
def wRange = rangePct > atrPct * 1.5;
def wVolume = volumeRatio >= 3;
def wBoring = boringness < 60;
def watchCount = wOn + wFiveDayHigh + wJump + wRange + wVolume + wBoring;

def statusCode =
    if !enoughHistory then 0
    else if blockedCount > 0 then 1
    else if watchCount > 0 then 2
    else 3;
def extraReasons = blockedCount + watchCount - 1;

plot StatusValue = statusCode;

AddLabel(yes,
    (if statusCode == 0 then "DATA? <21 BARS"
     else if statusCode == 1 then "BLOCKED " +
        (if gPrice then "PRICE<5"
         else if gVolume then "AVGVOL<500K"
         else if gWin then "WIN<60"
         else if gMean then "AVG<0.10"
         else if gFiveDay then "5D<=20D"
         else "BELOW SMA20")
     else if statusCode == 2 then "WATCH " +
        (if wOn then "ON>=3%"
         else if wFiveDayHigh then "5D>=5%"
         else if wJump then "5D JUMP"
         else if wRange then "RANGE>1.5ATR"
         else if wVolume then "VOL>=3X"
         else "BORING<60")
     else "CANDIDATE > MCC READY")
    + (if statusCode == 0 or statusCode == 3 then ""
       else if extraReasons == 1 then " +1"
       else if extraReasons == 2 then " +2"
       else if extraReasons >= 3 then " +3+"
       else ""),
    Color.BLACK);

AssignBackgroundColor(
    if statusCode == 3 then Color.GREEN
    else if statusCode == 2 then Color.YELLOW
    else if statusCode == 1 then Color.LIGHT_RED
    else Color.GRAY);
