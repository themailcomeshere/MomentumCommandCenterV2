# ============================================================
# AGGRESSIVE ENTRY LANE
# ============================================================
# Parallel orange lane. The normal green execution lane is unchanged.
# VWAP allowance is tiered by share price:
#   < $10    = 1.00% below VWAP
#   $10-$25  = 1.50% below VWAP
#   $25-$50  = 2.00% below VWAP
#   $50-$100 = 2.50% below VWAP
#   > $100   = 3.00% below VWAP
# The aggressive lane still blocks EOD, major failure, confirmed weakness,
# structural damage, and excessive ATR extension.
# ============================================================
input aggressiveMarginUnder10 = 1.00;
input aggressiveMargin10To25 = 1.50;
input aggressiveMargin25To50 = 2.00;
input aggressiveMargin50To100 = 2.50;
input aggressiveMarginOver100 = 3.00;
input requireMomentumAcceleration = yes;
input extensionWarningATR = 1.50;
input extensionDangerATR = 2.00;
# Fixed Arizona session cutoff for this version.
# The Eastern equivalents below are separate compile-time inputs because
# SecondsFromTime()/SecondsTillTime() require constant HHMM arguments.
def eodHourArizona = 15;
def eodMinuteArizona = 55;
input eodTimeEasternDaylight = 1855;
input eodTimeEasternStandard = 1955;
# ============================================================
# SESSION / TIMEZONE
#
# TOS chart display can remain set to Arizona time.
# thinkScript's SecondsFromTime() and SecondsTillTime() use the
# Eastern clock regardless of the chart's displayed timezone.
#
# Arizona remains UTC-7 year-round. Eastern is UTC-4 during US
# daylight time and UTC-5 during standard time. Therefore an
# Arizona 3:55 PM EOD corresponds to 6:55 PM Eastern during
# daylight time and 7:55 PM Eastern during standard time.
#
# The DST dates are calculated from the US rules:
#   - starts on the second Sunday in March
#   - ends on the first Sunday in November
#
# GetDayOfWeek() returns Monday=1 through Sunday=7.
# ============================================================
def currentDate =
    GetYYYYMMDD();
def currentYear =
    Floor(currentDate / 10000);
def currentMonth =
    Floor(currentDate / 100) % 100;
def currentDay =
    currentDate % 100;
def marchFirst =
    currentYear * 10000 + 301;
def novemberFirst =
    currentYear * 10000 + 1101;
def marchFirstDayOfWeek =
    GetDayOfWeek(marchFirst);
def novemberFirstDayOfWeek =
    GetDayOfWeek(novemberFirst);
def secondSundayMarch =
    8 + (7 - marchFirstDayOfWeek) % 7;
def firstSundayNovember =
    1 + (7 - novemberFirstDayOfWeek) % 7;
def easternDaylightTime =
    currentMonth > 3 and
    currentMonth < 11
    or
    currentMonth == 3 and
    currentDay >= secondSundayMarch
    or
    currentMonth == 11 and
    currentDay < firstSundayNovember;
def eodTimeEastern =
    if easternDaylightTime
    then eodTimeEasternDaylight
    else eodTimeEasternStandard;
# thinkScript requires SecondsTillTime()/SecondsFromTime() to receive
# compile-time constant HHMM arguments. The two Eastern times are therefore
# inputs (compile-time constants) and the DST branch selects between them.
def beforeEOD =
    if easternDaylightTime
    then SecondsTillTime(eodTimeEasternDaylight) > 0
    else SecondsTillTime(eodTimeEasternStandard) > 0;
def afterEOD =
    if easternDaylightTime
    then SecondsFromTime(eodTimeEasternDaylight) >= 0
    else SecondsFromTime(eodTimeEasternStandard) >= 0;
def regularSession =
    GetTime() >= RegularTradingStart(GetYYYYMMDD()) and
    GetTime() < RegularTradingEnd(GetYYYYMMDD());
def extendedSession =
    !regularSession;
def eodExitWindow =
    afterEOD;
# ============================================================
# TIME DIAGNOSTICS
#
# AddLabel() uses values from the last real bar, so these labels
# are intentionally supplemented by optional per-bar bubbles.
# The bubbles are attached to each historical bar and therefore
# follow the bars you zoom into.
# ============================================================
def secondsFromEOD =
    if easternDaylightTime
    then SecondsFromTime(eodTimeEasternDaylight)
    else SecondsFromTime(eodTimeEasternStandard);
def secondsTillEOD =
    if easternDaylightTime
    then SecondsTillTime(eodTimeEasternDaylight)
    else SecondsTillTime(eodTimeEasternStandard);
AddLabel(
    showDebugLabels and showTimeDiagnostics,
    "TIME AZ EOD=" + eodHourArizona + ":" + eodMinuteArizona +
    " ET=" + eodTimeEastern +
    " DST=" + easternDaylightTime
);
AddLabel(
    showDebugLabels and showTimeDiagnostics,
    "TIME CurrentDate=" + currentDate +
    " EODWindow=" + eodExitWindow
);
AddChartBubble(
    showTimeDiagnosticBubbles,
    high,
    "AZ=" + eodHourArizona + ":" + eodMinuteArizona +
    " ET=" + eodTimeEastern +
    " From=" + secondsFromEOD +
    " Till=" + secondsTillEOD +
    " EOD=" + eodExitWindow,
    if eodExitWindow then Color.RED else Color.WHITE,
    yes
);

# 1M PRICE / INDICATORS
# ============================================================
def price = close;
def ema9 = ExpAverage(price, fastEMA);
def ema20 = ExpAverage(price, slowEMA);
def ema9Rising = ema9 > ema9[1];
def ema20Rising = ema20 > ema20[1];
def ema9Falling = ema9 < ema9[1];
def ema20Falling = ema20 < ema20[1];
def dailyVWAP =
    vwap(period = AggregationPeriod.DAY);
def priceAboveVWAP = price > dailyVWAP;
def priceBelowVWAP = price < dailyVWAP;
def rsi =
    RSI(
        price = price,
        length = rsiLength
    );
def rsiRising = rsi > rsi[1];
def rsiFalling = rsi < rsi[1];
def rsiRecovering = rsi > rsi[1] and rsi[1] <= rsi[2];
def rsiWeakening = rsi < rsi[1] and rsi[1] <= rsi[2];
def macdValue =
    ExpAverage(price, macdFast) -
    ExpAverage(price, macdSlow);
def macdSignal =
    ExpAverage(macdValue, macdLength);
def macdHist = macdValue - macdSignal;
def macdBullish = macdValue > macdSignal;
def macdBearish = macdValue < macdSignal;
def macdRising = macdHist > macdHist[1];
def macdFalling = macdHist < macdHist[1];
def macdRecovering = macdHist > macdHist[1] and macdHist[1] <= macdHist[2];
def averageVolume = Average(volume, rvolLength);
def rvol =
    if averageVolume > 0
    then volume / averageVolume
    else 0;
def rvolHealthy = rvol >= 1.00;
def rvolStrong = rvol >= 1.50;
def atr =
    Average(
        TrueRange(
            high,
            close,
            low
        ),
        14
    );
def extensionATR =
    if atr > 0
    then AbsValue(price - dailyVWAP) / atr
    else 0;
def extensionWarning = extensionATR >= extensionWarningATR;
def extensionDanger = extensionATR >= extensionDangerATR;
def bullishExtensionDanger = price > dailyVWAP and extensionDanger;
# ============================================================
# 5M SECONDARY DATA
#
# 1M primary -> 5M secondary is valid in thinkScript.
# ============================================================
def fiveClose = close(period = AggregationPeriod.FIVE_MIN);
def fiveHigh = high(period = AggregationPeriod.FIVE_MIN);
def fiveLow = low(period = AggregationPeriod.FIVE_MIN);
def fiveOpen = open(period = AggregationPeriod.FIVE_MIN);
def fiveVolume = volume(period = AggregationPeriod.FIVE_MIN);
def fiveEMA9 = ExpAverage(fiveClose, fastEMA);
def fiveEMA20 = ExpAverage(fiveClose, slowEMA);
def fiveEMA9Rising = fiveEMA9 > fiveEMA9[1];
def fiveSMA50 = Average(fiveClose, fiveMinuteTrendSMA);
# The reconstructed 5M layer uses 5M price/EMA structure.
# Daily VWAP is handled separately by the 1M execution layer
# to avoid mixing two different secondary aggregations in one
# variable.
def fiveBullishEMAStack =
    fiveClose > fiveEMA9 and
    fiveEMA9 > fiveEMA20 and
    fiveEMA20 > fiveSMA50;
def fiveBullishCore =
    fiveEMA9 > fiveEMA20 and
    fiveClose > fiveEMA9;
def fiveBearishStack =
    fiveClose < fiveEMA9 and
    fiveEMA9 < fiveEMA20 and
    fiveEMA20 < fiveSMA50;
def fiveBearishCore =
    fiveEMA9 < fiveEMA20 and
    fiveClose < fiveEMA9;
# ============================================================
# 5M RSI / MACD / RVOL
# ============================================================
def fiveRSI =
    RSI(
        price = fiveClose,
        length = rsiLength
    );
def fiveRSIRising = fiveRSI > fiveRSI[1];
def fiveRSIFalling = fiveRSI < fiveRSI[1];
def fiveRSIWeakening = fiveRSI < fiveRSI[1] and fiveRSI[1] <= fiveRSI[2];
def fiveRSIRecovering = fiveRSI > fiveRSI[1] and fiveRSI[1] <= fiveRSI[2];
def fiveMACDValue =
    ExpAverage(fiveClose, macdFast) -
    ExpAverage(fiveClose, macdSlow);
def fiveMACDSignal =
    ExpAverage(fiveMACDValue, macdLength);
def fiveMACDHist =
    fiveMACDValue - fiveMACDSignal;
def fiveMACDBullish = fiveMACDValue > fiveMACDSignal;
def fiveMACDBearish = fiveMACDValue < fiveMACDSignal;
def fiveMACDRising = fiveMACDHist > fiveMACDHist[1];
def fiveMACDFalling = fiveMACDHist < fiveMACDHist[1];
def fiveMACDWeakening = fiveMACDHist < fiveMACDHist[1] and fiveMACDHist[1] < fiveMACDHist[2];
def fiveMACDRecovering = fiveMACDHist > fiveMACDHist[1] and fiveMACDHist[1] <= fiveMACDHist[2];
def fiveAverageVolume = Average(fiveVolume, rvolLength);
def fiveRVOL =
    if fiveAverageVolume > 0
    then fiveVolume / fiveAverageVolume
    else 0;
def fiveRVOLHealthy = fiveRVOL >= 1.00;
def fiveRVOLStrong = fiveRVOL >= 1.50;
# ============================================================
# 5M SETUP / BREAKOUT / RECLAIM / RETEST
# ============================================================
def fivePriorHigh = Highest(fiveHigh[1], breakoutLength);
def fivePriorLow = Lowest(fiveLow[1], breakoutLength);
def fiveBreakout = fiveClose > fivePriorHigh;
def fiveEMA9Reclaim =
    fiveClose > fiveEMA9 and
    fiveClose[1] <= fiveEMA9[1];
def fiveRetestEMA9 =
    fiveLow <= fiveEMA9 and
    fiveClose > fiveEMA9 and
    fiveClose > fiveOpen;
# For the 5M reconstruction, current 1M price relative to
# Daily VWAP is used only as an execution-layer context filter.
# The actual 5M chart remains the authoritative VWAP source.
# IMPORTANT DISTINCTION:
#   fiveSetupCandidate is a structural/setup event, not the permission
#   latch. The reconstructed fiveState below currently prioritizes this
#   setup before the PREPARE state, so a "5M BUY" bubble emitted by this
#   1M study can appear while fivePermission / fiveExecutionPermission
#   is false. Treat it as a reconstructed setup marker, NOT proof that
#   the actual 5M Command Center granted executable permission.
def fiveSetupCandidate =
    (
        fiveBreakout or
        fiveEMA9Reclaim or
        fiveRetestEMA9
    )
    and fiveBullishCore
    and fiveRSI >= minimumPermissionRSI
    and fiveRVOL >= minimumSetupRVOL;
# ============================================================
# 5M PERMISSION
#
# The 1M execution layer requires both reconstructed 5M
# structure and current 1M price to remain above Daily VWAP.
# ============================================================
def fiveBullishPermissionRaw =
    fiveBullishCore and
    fiveRSI >= minimumPermissionRSI and
    !fiveBearishStack;
def fiveStructuralDamage =
    (
        fiveClose < fiveEMA9 and
        fiveEMA9 < fiveEMA20
    )
    or
    (
        fiveClose < fiveEMA20 and
        fiveEMA9 < fiveEMA20 and
        !fiveEMA9Rising
    );
def fiveStructuralDamagePersistent =
    Sum(
        if fiveStructuralDamage then 1 else 0,
        permissionFailureBars
    ) >= permissionFailureBars;
def fiveMajorFailureRaw =
    fiveClose < fiveEMA9 and
    fiveEMA9 < fiveEMA20 and
    fiveEMA20 < fiveSMA50;
def fiveMajorFailurePersistent =
    Sum(
        if fiveMajorFailureRaw then 1 else 0,
        majorFailureBars
    ) >= majorFailureBars;
def fivePermissionState =
    CompoundValue(
        1,
        if eodExitWindow then
            0
        else if fiveMajorFailurePersistent then
            0
        else if fiveStructuralDamagePersistent then
            0
        else if fiveBullishPermissionRaw then
            1
        else
            fivePermissionState[1],
        0
    );
def fivePermission = fivePermissionState == 1;
# ============================================================
# 5M RUNNER / WANING / SWING ZONE
# ============================================================
def fiveRunnerStructure =
    fivePermission and
    fiveBullishEMAStack and
    priceAboveVWAP;
def fiveRunnerStrength =
    fiveRunnerStructure and
    (
        fiveRSI >= rsiStrong or
        fiveMACDBullish
    );
def fiveRunnerRising =
    fiveRunnerStrength and
    (
        fiveRSIRising or
        fiveMACDRising or
        fiveRVOLStrong
    );
# ============================================================
# TARGETED V14.4.5 CHANGE
#
# Keep the 5M runner-waning logic synchronized with V4.1.7. The
# prior test stopped evaluating waning as soon as price lost EMA9.
# The revised context keeps the bullish regime above Daily VWAP and
# EMA9>EMA20, while allowing price to dip below EMA9 so an actual
# reversal can be detected before full structural failure.
# This is a warning layer, not a SELL condition.
# ============================================================
def fiveRunnerWaningContext =
    fivePermission and
    priceAboveVWAP and
    fiveEMA9 > fiveEMA20 and
    !fiveMajorFailurePersistent;

def fiveRunnerWaningRaw =
    fiveRunnerWaningContext and
    (
        fiveRSIWeakening or
        fiveMACDWeakening or
        (
            fiveClose < fiveEMA9 and
            (
                fiveRSI < rsiStrong or
                fiveMACDFalling
            )
        ) or
        (
            fiveClose < fiveClose[1] and
            !fiveEMA9Rising and
            fiveRSIFalling
        )
    );

def fiveRunnerWaningPersistent =
    Sum(
        if fiveRunnerWaningRaw then 1 else 0,
        runnerWaningBars
    ) >= runnerWaningBars;
def fiveRunnerWaningEvent =
    fiveRunnerWaningPersistent and
    !fiveRunnerWaningPersistent[1];
def fiveSwingStructureIntact =
    fivePermission and
    fiveClose > fiveEMA20 and
    fiveEMA20 >= fiveSMA50 and
    priceAboveVWAP;
def fiveSwingMomentumWeakening =
    (
        fiveRSIWeakening and
        fiveMACDWeakening
    )
    or
    (
        fiveRSI < rsiStrong and
        fiveMACDFalling
    );
def fiveSwingZoneRaw =
    fiveSwingStructureIntact and
    fiveRunnerWaningPersistent and
    fiveSwingMomentumWeakening and
    !fiveMajorFailurePersistent;
def fiveSwingBuyEnvironment =
    fiveSwingZoneRaw and
    (
        fiveRSIRecovering or
        fiveMACDRecovering
    );
# ============================================================
# 5M FAILURE STATES
# ============================================================
def fivePrepareToSell =
    (
        fiveClose < fiveEMA20 and
        !fiveBullishCore
    )
    or
    fiveMajorFailurePersistent;
def fiveConfirmedWeakness =
    fiveClose < fiveEMA9 and
    fiveEMA9 < fiveEMA20 and
    (
        fiveRSI < rsiWeak or
        fiveMACDBearish
    );
def fiveBreakdown =
    fiveMajorFailurePersistent and
    fiveConfirmedWeakness and
    fiveClose < fiveSMA50;

# ============================================================
# 5M PARALLEL AGGRESSIVE ENVIRONMENT
# ============================================================
# This reconstructs only the 5M conditions needed by the orange lane.
# It does not modify fivePermission, fiveState, or any green execution gate.
# Daily VWAP remains the 1M execution layer's authoritative VWAP.
# ============================================================
def aggressiveVWAPMarginPct =
    if price < 10 then aggressiveMarginUnder10
    else if price < 25 then aggressiveMargin10To25
    else if price < 50 then aggressiveMargin25To50
    else if price < 100 then aggressiveMargin50To100
    else aggressiveMarginOver100;
def aggressiveVWAPFloor =
    dailyVWAP * (1 - aggressiveVWAPMarginPct / 100);
def aggressivePriceWithinVWAPMargin =
    price >= aggressiveVWAPFloor;
def fiveAggressiveStructure =
    fiveClose > fiveEMA20 and
    fiveClose > fiveEMA9 and
    fiveEMA9 >= fiveEMA20;
def fiveAggressiveMomentum =
    fiveRSI >= minimumPermissionRSI and
    (
        fiveMACDBullish or
        fiveRSIRising or
        fiveRVOLHealthy
    );
def fiveAggressiveEnvironment =
    aggressivePriceWithinVWAPMargin and
    fiveAggressiveStructure and
    fiveAggressiveMomentum and
    !fiveStructuralDamagePersistent and
    !fiveMajorFailurePersistent and
    !fiveConfirmedWeakness and
    !eodExitWindow and
    !bullishExtensionDanger;
def fiveAggressiveTrigger =
    fiveSetupCandidate or
    (
        fiveClose > fiveClose[1] and
        fiveRSIRising and
        (fiveMACDBullish or fiveMACDRising)
    );
# A 5M setup candidate is not a reason to disable the aggressive lane.
# It is already one of the setup/trigger signals and can occur while the
# normal execution lane remains unavailable because price is below VWAP.
# Keep the pre-VWAP allowance and all structural/failure/extension guards.
def fiveAggressiveLaneActive =
    fiveAggressiveEnvironment and
    !priceAboveVWAP and
    !fiveRunnerStructure;

# 5M STATE CODES
# ============================================================
def FIVE_WAIT = 0;
def FIVE_PREPARE_BUY = 1;
def FIVE_BUY = 2;
def FIVE_RUNNER_RISING = 3;
def FIVE_RUNNER = 4;
def FIVE_RUNNER_WANING = 5;
def FIVE_SWING_ZONE = 6;
def FIVE_PREPARE_SELL = 7;
def FIVE_SELL_REBOUND = 8;
def FIVE_SELL_CONFIRMED = 9;
def FIVE_SELL_BREAKDOWN = 10;
def FIVE_EOD_GET_OUT = 11;
# This is the 1M study's reconstructed 5M state machine. It is not a
# direct read of the separate 5M study's state variable. Compare it with
# the actual 5M chart and the companion Authority Diagnostic when auditing.
def fiveRawState =
    if eodExitWindow then
        FIVE_EOD_GET_OUT
    else if fiveBreakdown then
        FIVE_SELL_BREAKDOWN
    else if fiveConfirmedWeakness then
        FIVE_SELL_CONFIRMED
    else if fivePrepareToSell then
        FIVE_PREPARE_SELL
    else if fiveSwingZoneRaw then
        FIVE_SWING_ZONE
    else if fiveRunnerWaningPersistent then
        FIVE_RUNNER_WANING
    else if fiveRunnerRising then
        FIVE_RUNNER_RISING
    else if fiveRunnerStructure then
        FIVE_RUNNER
    else if fiveSetupCandidate then
        FIVE_BUY
    else if fivePermission then
        FIVE_PREPARE_BUY
    else
        FIVE_WAIT;
def fiveState = fiveRawState;

def priorOneMHigh = Highest(high[1], breakoutLength);
def oneMBreakout = price > priorOneMHigh;
def oneMVWAPReclaim =

def oneMEMA9Reclaim =
    price > ema9 and
    price[1] <= ema9[1];

    price > ema9 and
    price > open;
def oneMVWAPRetest =
    low <= dailyVWAP and
    price > dailyVWAP and
    price > open;
def freshTrigger =
    oneMBreakout or
    oneMVWAPReclaim or
    oneMEMA9Reclaim or
    oneMEMA9Retest or
    oneMVWAPRetest;

# 1M MOMENTUM
# ============================================================
def momentumBullish =
    price > ema9 and
    ema9 > ema20 and
    rsi >= minimumExecutionRSI and
    macdBullish;
def momentumAccelerating =
    (
        rsiRising and
        macdRising
    )
    or
    (
        rsiRising and
        rvolHealthy
    )
    or
    (
        macdRising and

    price > open and
    (
        if high > low
        then (price - low) / (high - low) >= 0.60
        else no
    );

# EXECUTION FILTERS

def fiveMinuteHealthy =
    fivePermission and
    !fiveMajorFailurePersistent and
    !fiveConfirmedWeakness;
# Canonical 5M execution gate. Every 1M execution path uses this
# same gate so the candidate, READY logic, and diagnostics cannot
# disagree about whether the 5M layer currently permits execution.
def fiveExecutionPermission =
    fivePermission and
    priceAboveVWAP and
    !fiveMajorFailurePersistent and
    !fiveConfirmedWeakness;
def notFiveSwingSell =
    !fiveSwingZoneRaw;
def executionVolume =
    rvol >= minimumExecutionRVOL;
def executionStructure =
    fiveExecutionPermission and
    !bullishExtensionDanger;
# ============================================================
# PREPARE TO BUY
# ============================================================
# Runner waning invalidates the normal prepare-to-buy pathway.
# A swing reclaim is handled separately below.
def prepareToBuyEnvironment =
    fiveExecutionPermission and
    !fiveSwingZoneRaw and
    priceAboveVWAP and
    price > ema20 and
    ema20Rising and
    rsi >= rsiBull;
def prepareToBuyInvalidated =
    eodExitWindow
    or fiveMajorFailurePersistent
    or fiveConfirmedWeakness
    or fiveRunnerWaningPersistent
    or fiveSwingZoneRaw
    or price < dailyVWAP
    or price < ema20
    or rsi < rsiWeak;
rec prepareToBuyState =
    if eodExitWindow then
        0
    else if prepareToBuyInvalidated then
        0
    else if prepareToBuyEnvironment then
        1
    else
        prepareToBuyState[1];
def prepareToBuy = prepareToBuyState == 1;
def newPrepareToBuy =
    prepareToBuy and
    !prepareToBuy[1];
# ============================================================
# EXECUTION CANDIDATE ENVIRONMENT
# Candidate creation is intentionally broader than final READY.
# It identifies an actionable 1M opportunity inside valid 5M
# permission without requiring final candle/RVOL/acceleration
# confirmation at the candidate stage.
# ============================================================
def executionCandidateEnvironment =
    fiveExecutionPermission and
    !fiveSwingZoneRaw and
    priceAboveVWAP and
    price > ema20 and
    ema20Rising;
def executionCandidateMomentum =
    rsi >= minimumExecutionRSI and
    macdBullish and
    (
        price > ema9 or
        price > price[1] or
        rsiRising or
        macdRising or
        rvolHealthy
    );
def executionCandidate =
    executionCandidateEnvironment and
    executionCandidateMomentum;
# ============================================================
# 1M CANDIDATE CONFIRMATION EVENT
# Candidate confirmation is independent of freshTriggerEvent.
# It is a real 1M confirmation event occurring inside an
# already-valid execution candidate environment.
# ============================================================
def candidateConfirmationRaw =
    executionCandidate and
    qualityBullishCandle and
    (
        momentumAccelerating or
        price > high[1] or
        rsi >= rsiStrong
    ) and
    !bullishExtensionDanger;
def candidateConfirmationEvent =
    candidateConfirmationRaw and
    !candidateConfirmationRaw[1];
# ============================================================
# CONTINUATION: PAUSE / ROTATION DETECTION
# ============================================================
def continuationPauseEvidence =
    price <= price[1]
    or high <= high[1]
    or !rsiRising
    or !macdRising;
def continuationPauseCount =
    Sum(
        if continuationPauseEvidence then 1 else 0,
        continuationLookbackBars
    );
def continuationPauseDetected =
    continuationPauseCount >= continuationPauseBarsRequired;
# ============================================================
# CONTINUATION: BULLISH RESUMPTION
# ============================================================
def continuationResume =
    price > ema9 and
    ema9Rising and
    price > high[1] and
    rsi >= continuationMinimumRSI and
    macdBullish and
    (
        rsiRising or
        macdRising or
        rvolHealthy
    );
def continuationBaseEnvironment =
    fiveRunnerStructure and
    fiveExecutionPermission and
    !fiveSwingZoneRaw;
# ============================================================
# CONTINUATION: ARMED STATE
# ============================================================
rec continuationArmed =
    if eodExitWindow then
        0
    else if !continuationBaseEnvironment then
        0
    else if continuationResume then
        0
    else if continuationPauseDetected then
        1
    else
        continuationArmed[1];
def continuationExecutionEventRaw =
    continuationBaseEnvironment and
    continuationArmed[1] and
    continuationResume and
    executionVolume and
    qualityBullishCandle and
    !bullishExtensionDanger;
# ============================================================
# CONTINUATION RE-ARM / COOLDOWN
# ============================================================
rec continuationCooldownAge =
    if continuationExecutionEventRaw then
        0
    else if continuationCooldownAge[1] >= 0 then
        continuationCooldownAge[1] + 1
    else
        continuationCooldownBars;
def continuationCooldownReady =
    continuationCooldownAge[1] >= continuationCooldownBars;
def continuationExecutionEventQualified =
    continuationExecutionEventRaw and
    continuationCooldownReady;
# ============================================================
# CANDIDATE PAUSE / RE-ARM
# A pause means the candidate condition has cleared while the
# broader 5M execution environment remains intact. This is not
# a bearish reset. It allows a later reclaim/resumption to create
# a new execution opportunity without demanding full trend failure.
# ============================================================
def candidatePauseRaw =
    !executionCandidate and
    fiveExecutionPermission and
    priceAboveVWAP and
    price > ema20;
rec candidatePauseCount =
    if candidatePauseRaw then
        candidatePauseCount[1] + 1
    else
        0;
rec candidateEverActive =
    if executionCandidate then
        1
    else
        candidateEverActive[1];

def candidateRearmReady =
    !candidateEverActive[1] or
    candidatePauseCount >= candidateRearmBars;
# ============================================================
# FRESH / IN-TREND CANDIDATES
# ============================================================
def freshTriggerEvent =
    freshTrigger and
    !freshTrigger[1];
def freshExecutionCandidate =
    executionCandidate and
    freshTrigger;
def inTrendExecutionCandidate =
    executionCandidate and
    !freshTrigger;
# ============================================================
# EXECUTION OPPORTUNITY / EVENT GATING
# Candidate conditions may persist, but an execution window is
# created only by a new candidate event, including a genuine 1M
# confirmation event inside the valid candidate environment.
# This prevents a long bullish stretch from repeatedly holding
# the same opportunity open.
# ============================================================
def freshExecutionEvent =
    freshExecutionCandidate and
    freshTriggerEvent and
    candidateRearmReady[1];
def inTrendExecutionEvent =
    inTrendExecutionCandidate and
    !inTrendExecutionCandidate[1] and
    candidateRearmReady[1];
def continuationExecutionEvent =
    continuationExecutionEventQualified and
    !continuationExecutionEventQualified[1];
def executionCandidateEvent =
    freshExecutionEvent
    or inTrendExecutionEvent
    or continuationExecutionEvent
    or candidateConfirmationEvent;
def continuationCandidate = continuationExecutionEventRaw;
# ============================================================
# ACTIVE EXECUTION OPPORTUNITY
# ============================================================
def executionOpportunity =
    executionCandidateEvent;
# ============================================================
# EXECUTION AGE / FINITE WINDOW
# Bar 0 = trigger. Bar 1 = still actionable. Bar 2 = expired.
# ============================================================
def newExecutionTrigger =
    executionOpportunity and
    !executionOpportunity[1];
rec executionAge =
    if newExecutionTrigger then
        0
    else if executionAge[1] >= 0 and
            executionAge[1] < executionWindowBars then
        executionAge[1] + 1
    else
        -1;
def executionWindowActive =
    executionAge >= 0 and
    executionAge < executionWindowBars;
# ============================================================
# EXECUTION READY — CANONICAL 5M GATE
# ============================================================
def executionReady =
    executionWindowActive and
    fiveExecutionPermission and
    !bullishExtensionDanger;
def executionInvalidated =
    !fiveExecutionPermission
    or price < ema20
    or price < dailyVWAP
    or rsi < rsiWeak
    or macdBearish
    or bullishExtensionDanger;
def finalExecutionReady =
    executionReady and
    !executionInvalidated;

# ============================================================
# 1M AGGRESSIVE EXECUTION LANE
# ============================================================
# The orange lane is deliberately parallel to the normal green lane.
# It can operate while the normal VWAP gate is not yet satisfied, but
# it still requires a healthy 5M developing environment and 1M momentum.
# ============================================================
def aggressiveExecutionEnvironment =
    fiveAggressiveLaneActive and
    !fiveSwingZoneRaw and
    price > ema20 and
    ema20Rising and
    rsi >= minimumExecutionRSI and
    (
        price > ema9 or
        price > price[1] or
        rsiRising or
        macdRising or
        rvolHealthy
    ) and
    !bullishExtensionDanger and
    !eodExitWindow;
def aggressiveExecutionTrigger =
    oneMBreakout or
    oneMEMA9Reclaim or
    oneMEMA9Retest or
    (
        price > price[1] and
        rsiRising and
        (macdBullish or macdRising)
    );
def aggressiveExecutionOpportunity =
    aggressiveExecutionEnvironment and
    !finalExecutionReady;
def aggressivePrepareToBuy =
    aggressiveExecutionOpportunity and
    !aggressiveExecutionTrigger;
def aggressiveBuy =
    aggressiveExecutionOpportunity and
    aggressiveExecutionTrigger;
def aggressivePrepareEvent =
    aggressivePrepareToBuy and
    !aggressivePrepareToBuy[1];
def aggressiveBuyEvent =
    aggressiveBuy and
    !aggressiveBuy[1];

# ============================================================
# V14.4.8 AGGRESSIVE-LANE BLOCKER CODE (DIAGNOSTIC ONLY)
#
# This code does not alter the aggressive lane. It reports the first
# failed condition so historical bars can be audited without guessing.
#   0  = aggressive BUY condition is currently valid
#   1  = EOD exit window
#   2  = price below the price-tiered VWAP allowance floor
#   3  = reconstructed 5M structure not bullish enough
#   4  = reconstructed 5M momentum not sufficient
#   5  = persistent 5M structural damage
#   6  = persistent 5M major failure
#   7  = reconstructed 5M confirmed weakness
#   8  = bullish ATR-extension danger
#   9  = price already above Daily VWAP (aggressive lane is pre-VWAP)
#  10  = 5M setup candidate already active
#  11  = reconstructed 5M runner structure already active
#  12  = reconstructed 5M swing zone active
#  13  = 1M price not above EMA20
#  14  = 1M EMA20 not rising
#  15  = 1M RSI below minimum execution RSI
#  16  = no qualifying 1M momentum evidence
#  17  = aggressive environment valid but no fresh 1M trigger
#  18  = aggressive environment is not active for another reason
#  19  = normal execution READY already active; orange lane suppressed
# ============================================================
def aggressiveMomentumEvidence =
    price > ema9 or
    price > price[1] or
    rsiRising or
    macdRising or
    rvolHealthy;
def aggressiveBlockCode =
    if eodExitWindow then 1
    else if finalExecutionReady then 19
    else if !aggressivePriceWithinVWAPMargin then 2
    else if !fiveAggressiveStructure then 3
    else if !fiveAggressiveMomentum then 4
    else if fiveStructuralDamagePersistent then 5
    else if fiveMajorFailurePersistent then 6
    else if fiveConfirmedWeakness then 7
    else if bullishExtensionDanger then 8
    else if priceAboveVWAP then 9
    # Code 10 retired: fiveSetupCandidate may now activate the aggressive lane.
    else if fiveRunnerStructure then 11
    else if fiveSwingZoneRaw then 12
    else if price <= ema20 then 13
    else if !ema20Rising then 14
    else if rsi < minimumExecutionRSI then 15
    else if !aggressiveMomentumEvidence then 16
    else if !aggressiveExecutionTrigger then 17
    else if !aggressiveExecutionEnvironment then 18
    else 0;

def executionIsContinuation =
    continuationExecutionEventQualified and
    !freshExecutionCandidate;
def executionIsInTrend =
    inTrendExecutionCandidate and
    !freshExecutionCandidate and
    !continuationExecutionEventQualified;
def executionSignalType =
    if !finalExecutionReady then 0
    else if executionIsContinuation then 2
    else if executionIsInTrend then 3
    else 1;
# ============================================================
# EXECUTION BLOCK DIAGNOSTICS
#
# These codes are diagnostic only. They do not change execution
# behavior. They identify the first active blocker for READY and
# separately identify whether an execution opportunity exists.
#
# READY BLOCK CODE
#   0 = READY
#   1 = EOD
#   2 = no active execution opportunity
#   3 = execution window expired
#   4 = 5M execution permission failed
#   5 = swing zone active
#   6 = bullish extension danger
#   7 = price below EMA20
#   8 = price below Daily VWAP
#   9 = RSI below weak threshold
#  10 = MACD bearish
#
# CANDIDATE BLOCK CODE
#   0 = candidate environment active
#   1 = EOD
#   2 = 5M execution permission failed
#   3 = swing zone active
#   4 = price below Daily VWAP
#   5 = price not above EMA20
#   6 = EMA20 not rising
#   7 = RSI below minimum execution RSI
#   8 = bullish extension danger
#   9 = MACD not bullish
#  10 = no candidate momentum evidence
#
# These diagnostics are deliberately explicit so a chart can be
# evaluated without inferring a blocker from several labels.
# ============================================================
def executionReadyBlockCode =
    if eodExitWindow then
        1
    else if executionAge >= executionWindowBars and
            executionAge < executionWindowBars + 1 then
        3
    else if !executionWindowActive then
        2
    else if !fiveExecutionPermission then
        4
    else if fiveSwingZoneRaw then
        5
    else if bullishExtensionDanger then
        6
    else if price < ema20 then
        7
    else if price < dailyVWAP then
        8
    else if rsi < rsiWeak then
        9
    else if macdBearish then
        10
    else
        0;
def executionCandidateBlockCode =
    if eodExitWindow then
        1
    else if !fiveExecutionPermission then
        2
    else if fiveSwingZoneRaw then
        3
    else if !priceAboveVWAP then
        4
    else if price <= ema20 then
        5
    else if !ema20Rising then
        6
    else if rsi < minimumExecutionRSI then
        7
    else if bullishExtensionDanger then
        8
    else if !macdBullish then
        9
    else if !(
        price > ema9 or
        price > price[1] or
        rsiRising or
        macdRising or
        rvolHealthy
    ) then
        10
    else
        0;
# ThinkScript def variables are numeric/data-holder expressions.
# Keep executionCandidateBlockCode numeric and emit the human-readable
# block reason directly in the chart-bubble branches below.
# ============================================================
# HISTORICAL EXECUTION DIAGNOSTICS
# AddLabel() always reflects the last real bar. These optional bubbles

                else if inTrendExecutionEvent then "INTR"
                else "NONE") +
    " O=" + executionOpportunity +
    " R=" + finalExecutionReady +
    " AGE=" + executionAge +
    "\nEXT=" + Round(extensionATR, 2) +
    " BLK=" + executionReadyBlockCode +
    " CB=" + executionCandidateBlockCode +
    "\nRSI=" + Round(rsi, 1) +
    " RVOL=" + Round(rvol, 2) +
    " VWAP=" + priceAboveVWAP +
    " EMA20=" + (price > ema20),
    if finalExecutionReady then Color.GREEN
    else if executionOpportunity then Color.YELLOW
    else Color.WHITE,
    yes
);

    macdBearish;

    if eodExitWindow then
        EOD_GET_OUT
    else if sellConfirmed then
        SELL
    else if prepareToSell then
        PREPARE_SELL
    else if swingReclaimConfirmed then
        SWING_RECLAIM
    else if finalExecutionReady then
        EXECUTION_READY
    else if executionExpired then
        EXECUTION_EXPIRED
    else if prepareToBuy then
        PREPARE
    else
