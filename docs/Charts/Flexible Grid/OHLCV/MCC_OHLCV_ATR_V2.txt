# MCC_OHLCV_ATR_V2.1 — OHLCV + Thinkorswim ATR parity export
#
# STATUS: Aggregation selector added. Compile and validate each selected chart
# aggregation in Thinkorswim before relying on its output.
# PURPOSE: Versioned Strategy Report transport carrying the completed source
# chart bar's OHLCV plus Thinkorswim Wilder ATR for that exact source bar.
#
# IMPORTANT AGGREGATION BEHAVIOR:
# - Select the desired aggregation in the study's "chartAggregation" input.
# - ALSO set the Thinkorswim chart itself to the same aggregation using the
#   chart's aggregation control. A study cannot change its host chart's
#   aggregation; this input validates the selected chart period and controls
#   the payload's timeframe label. Mismatches are blocked and shown in red.
# - Supported periods: 1m, 5m, 15m, 30m, 1h, and 4h.
#
# SETUP:
# - Add this strategy to an EVR chart.
# - Choose chartAggregation below and set the chart to the same period.
# - Load sufficient history: at least 99 consecutive completed source bars
#   before the first comparison bar (98 Wilder prefetch bars + comparison bar).
# - Match chart extended-hours and corporate-action adjustment settings to MCC.
# - ATR is calculated on EVERY bar present in the host chart. Therefore the
#   export must also include EVERY bar used by that calculation. Do not filter
#   payload rows by time of day: doing so drops premarket/after-hours bars that
#   update Wilder ATR and creates an unreproducible state jump at the next
#   exported bar.
# - Set the chart's extended-hours display ON. The date range below limits the
#   export only by calendar date; all available chart bars within those dates
#   are exported. Choose a start date far enough ahead of the comparison window
#   to provide at least 99 earlier completed bars.
# - Filter to the strategy's eligible 07:00–19:59 ET window downstream, after
#   ATR parity has been calculated on the complete chart-bar sequence.
# - Export the Strategy Report and preserve full order names/precision.
#
# V2 payload:
# MCC_OHLCV_ATR|v2.1|symbol|timeframe|epochMillis|open|high|low|close|volume|atr
#
# ATR is calculated from chart bars using ThinkScript Wilder averaging.
# TrueRange receives the unshifted close series; the function itself references
# the previous close. Do not pass close[1], which shifts the previous-close
# reference back an additional bar. Index [1] exports the completed source bar.

declare upper;

input startDate = 20261019;
input endDate = 20261019;
input chartAggregation = AggregationPeriod.FIVE_MIN;
input atrLength = 14;

def chartAggregationMatches = GetAggregationPeriod() == chartAggregation;
def aggregationIsSupported =
    chartAggregation == AggregationPeriod.MIN or
    chartAggregation == AggregationPeriod.FIVE_MIN or
    chartAggregation == AggregationPeriod.FIFTEEN_MIN or
    chartAggregation == AggregationPeriod.THIRTY_MIN or
    chartAggregation == AggregationPeriod.HOUR or
    chartAggregation == AggregationPeriod.FOUR_HOURS;

def sourceDate = GetYYYYMMDD()[1];
def sourceInDateRange = sourceDate >= startDate and sourceDate <= endDate;
def sourceTimestamp = GetTime()[1];
def sourceOpen = open[1];
def sourceHigh = high[1];
def sourceLow = low[1];
def sourceClose = close[1];
def sourceVolume = volume[1];

def trueRange = TrueRange(high, close, low);
def atrSeries = MovingAverage(AverageType.WILDERS, trueRange, atrLength);
def sourceAtr = atrSeries[1];

# Export all chart bars in the selected date range. ATR uses the unfiltered
# host-chart TrueRange series, so applying a time-of-day export filter here
# would omit bars that still affect the Wilder recurrence.
def exportBar = BarNumber() > 2 and aggregationIsSupported
    and chartAggregationMatches
    and sourceInDateRange
    and !IsNaN(sourceAtr);

plot MCCExportMarker = if exportBar then low else Double.NaN;
MCCExportMarker.SetPaintingStrategy(PaintingStrategy.POINTS);
MCCExportMarker.SetDefaultColor(Color.CYAN);
MCCExportMarker.SetLineWeight(3);

AddLabel(yes,
    "MCC_OHLCV_ATR_V2 | selected=" +
    (if chartAggregation == AggregationPeriod.MIN then "1m"
     else if chartAggregation == AggregationPeriod.FIVE_MIN then "5m"
     else if chartAggregation == AggregationPeriod.FIFTEEN_MIN then "15m"
     else if chartAggregation == AggregationPeriod.THIRTY_MIN then "30m"
     else if chartAggregation == AggregationPeriod.HOUR then "1h"
     else if chartAggregation == AggregationPeriod.FOUR_HOURS then "4h"
     else "unsupported") +
    " | chart=" + (if chartAggregationMatches then "MATCH" else "MISMATCH") +
    " | ATR(" + atrLength + ") WILDERS",
    if aggregationIsSupported and chartAggregationMatches then Color.CYAN else Color.RED);
AddLabel(yes,
    if aggregationIsSupported and chartAggregationMatches
    then "Full chart-history export enabled — record ETH/adjustment settings"
    else "EXPORT BLOCKED — select a supported aggregation and match the chart",
    if aggregationIsSupported and chartAggregationMatches then Color.YELLOW else Color.RED);

AddOrder(OrderType.BUY_TO_OPEN, exportBar,
    close,
    1,
    Color.CURRENT,
    Color.CURRENT,
    "MCC_OHLCV_ATR|v2.1|" + GetSymbol() + "|" +
    (if chartAggregation == AggregationPeriod.MIN then "1m"
     else if chartAggregation == AggregationPeriod.FIVE_MIN then "5m"
     else if chartAggregation == AggregationPeriod.FIFTEEN_MIN then "15m"
     else if chartAggregation == AggregationPeriod.THIRTY_MIN then "30m"
     else if chartAggregation == AggregationPeriod.HOUR then "1h"
     else if chartAggregation == AggregationPeriod.FOUR_HOURS then "4h"
     else "unsupported") + "|" +
    sourceTimestamp + "|" + sourceOpen + "|" + sourceHigh + "|" +
    sourceLow + "|" + sourceClose + "|" + sourceVolume + "|" + sourceAtr);

AddOrder(OrderType.SELL_TO_CLOSE, exportBar,
    close,
    1,
    Color.CURRENT,
    Color.CURRENT,
    "MCC_OHLCV_ATR_TRANSPORT_CLOSE");

# VALIDATION:
# 1. Compile in Thinkorswim for each selected aggregation.
# 2. Set chartAggregation input AND chart aggregation control to same period.
# 3. Confirm the label shows chart=MATCH and export is enabled.
# 4. Set a sufficiently early startDate to include >=99 completed bars before
#    the first comparison bar. Do not restrict export by time of day.
# 5. Confirm premarket, regular-session and after-hours chart bars are exported
#    throughout the selected date range.
# 6. Compare >=20 consecutive exported bars with chart OHLCV and ATR plot.
# 7. Record chart timezone, ETH, adjustment, ATR length and average type.
# 8. Missing source timestamps remain gaps; never interpolate them.
# 9. Apply the 07:00–19:59 ET sell-window filter only after ATR parity.
